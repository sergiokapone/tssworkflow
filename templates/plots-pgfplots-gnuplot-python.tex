%=================== Побудова засобами pgfplots ===================
\begin{tikzpicture}
	\begin{axis}[cartesian]
		\addplot+[blue, no marks, domain={-6:6}] {x^2};
	\end{axis}
\end{tikzpicture}

\begin{tikzpicture}
	\begin{axis}[scientific]
		\addplot [only marks, samples = 50, red, thick] {x^3};
	\end{axis}
\end{tikzpicture}
\\%============= Побудова засобами pgfplots+gnuplot ===============
\begin{tikzpicture}
	\begin{axis}[cartesian]
		\addplot [red, no marks, raw gnuplot] gnuplot {
		plot [x=-6:6] x**2;
		};
	\end{axis}
\end{tikzpicture}
\\%================ Побудова засобами gnuplottex ===================
\begin{gnuplot}[terminal=epslatex]
    set grid ytics lc rgb "gray" lw 1 lt 0
    set grid xtics lc rgb "gray" lw 1 lt 0
       
    set xtics axis

    set ytics axis
    set zeroaxis linewidth 2 linetype 1 lc "black"
    set border 0
    unset key
    
	plot [x=-6:6] x**2 lc "red"
\end{gnuplot}
\\%================== Побудова засобами python =====================
\begin{pycode}
import numpy as np
import matplotlib.pyplot as plt
x = np.linspace(-6, 6, 1000)

plt.xlabel('x')
plt.ylabel('y')
plt.grid()
ax = plt.gca()
for spine in ['left', 'bottom']:
    ax.spines[spine].set_position('zero')

# Hide the other spines...  
for spine in ['right', 'top']:
    ax.spines[spine].set_color('none')

plt.plot(x, x**2, 'r')

plt.savefig('myplot.pdf')
print (r'\includegraphics[width=\linewidth]{myplot.pdf}')
\end{pycode}
