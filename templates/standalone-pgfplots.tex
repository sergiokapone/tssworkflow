% !TSS name: Standalone: pgfplots (графік)
% !TSS description: \documentclass[tikz]{standalone} + pgfplots: коло через параметричний \addplot (кут у градусах, domain=0:360), рівні масштаби осей; виправлена версія шаблону з крапкою з комою
% !TeX program = pdflatex
\documentclass[tikz, border=${ask:border:5pt}]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}
	\begin{axis}[
		axis equal,
		axis lines = middle,
		xlabel = {$x$},
		ylabel = {$y$},
		grid = both,
	]
		\addplot[blue, thick, domain=0:360, samples=100] ({cos(x)}, {sin(x)});${cursor}
	\end{axis}
\end{tikzpicture}
\end{document}
